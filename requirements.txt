# Runtime dependencies for the Railway service (the API).
#
# ML stack pinned to the training environment. Verified 28 Aug 2026: these
# pins load all six joblib artifacts and produce BIT-IDENTICAL predictions on
# Python 3.9 and 3.12, so the runtime may use either. Do not bump without
# re-exporting the artifacts from research/03_models.ipynb.
#
# Python is pinned to 3.12 in .python-version. That is the CEILING, not a
# preference: numpy 1.26.4 and pandas 2.1.4 publish no cp313 wheels, so 3.13
# would fall back to building from source (or fail).
scikit-learn==1.6.1
# xgboost-cpu, NOT xgboost: plain xgboost resolves to the ~224MB CUDA wheel on
# Linux, which is pure weight on a CPU-only container. Note this package ships
# Linux/Windows wheels only -- macOS has none, which is why the local file
# (backend/requirements.txt) pins plain xgboost instead.
xgboost-cpu==2.1.4
lightgbm==4.6.0
joblib==1.5.2
numpy==1.26.4
pandas==2.1.4

# Web stack, pinned for the same reason the ML stack above is: a rebuild
# months from now must install what is running today. These were floors
# (`>=`) until 2026-08-29, which meant any redeploy could silently pull a new
# major -- an odd risk to carry directly underneath artifacts pinned to the
# digit. Verified to resolve and serve on Python 3.12 (Railway's version).
#
# starlette and pydantic are transitive through FastAPI, but they are what
# actually implement the CORS middleware, routing and request validation this
# API depends on -- and FastAPI's own bound is `pydantic>=2.7.0` with no
# ceiling, so leaving them unpinned leaves the real surface unpinned.
fastapi==0.128.0
starlette==0.49.3
pydantic==2.12.5
uvicorn==0.39.0
sqlalchemy==2.0.52
psycopg2-binary==2.9.12
